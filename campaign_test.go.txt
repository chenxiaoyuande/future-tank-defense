 campaignSetup(1,true)

func TestCampaignFiniteWavesAndBossProgression(t *testing.T) {
 for n:=1;n<=3;n++ {
  campaignSetup(n,true);shield=100000;shield2=100000
  for tick:=0;tick<1800 && !bossStarted;tick++ {
   for i:=2;i<191;i++ {if entity[i]==2 {damageZombie(i,99)};if entity[i]==7 {entity[i]=0}}
   updateGame()
  }
  if !bossStarted || stageKills!=waveTarget || waveSpawned!=waveTarget {t.Fatal("wave not finite",n,stageKills,waveSpawned)}
  if zombieType[bossSlot]!=3 || hp[bossSlot]!=22+n*6 {t.Fatal("boss scaling")}
  damageZombie(bossSlot,999);updateGame()
  want:=2;if n==3 {want=3};if state!=want {t.Fatal("clear or final victory missing",n,state)}
  if n<3 {old:=score;nextLevel();if level!=n+1 || lives!=3 || lives2!=3 || score!=old || weaponLevel!=1 {t.Fatal("next stage progression")}}
 }
}

func TestCampaignMenuPauseAndRestart(t *testing.T) {
 campaignSetup(1,false);state=1;menuDown=false;confirmDown=false
 keys[Key2]=true;keys[KeyH]=true;updateControls()
 if !twoPlayers || selectedLevel!=3 || state!=1 {t.Fatal("selection failed")}
 keys[KeyEnter]=true;updateControls();if level!=3 || state!=0 || entity[191]!=1 {t.Fatal("start failed")}
 paused=true;before:=ticks;cool:=ultimateCooldown2;updateGame()
 if ticks!=before || ultimateCooldown2!=cool {t.Fatal("pause advanced")}
 paused=false;keys=map[int]bool{KeyM:true};updateControls();before=ticks;updateGame()
 if state!=1 || ticks!=before {t.Fatal("menu did not stop combat")}
}
func TestCooperativeFriendlyFire(t *testing.T) {
 campaignSetup(1,true)
 for i:=2;i<191;i++ {entity[i]=0}
 px[1]=-100;py[1]=0;px[191]=0;py[191]=0;shield2=0

}
