func TestHealthAndDeathOnce(t *testing.T) {
 for kind,total:=range []int{5,1,3} {

}

type Item struct {ID,Kind,Facing,Age,Type,HP,Growth,Windup,PickupType int; X,Y,VX,VY,Dash float64; Rocket,Fire bool}
type Snapshot struct {Label string;Tick,Score,Kills,Lives,Shield,ShotDelay,Cooldown,Weapon,Nut,Speed,Pepper int;Entities []Item}
func snapshot(label string) Snapshot {
 s:=Snapshot{Label:label,Tick:ticks,Score:score,Kills:kills,Lives:lives,Shield:shield,ShotDelay:shotDelay,Cooldown:ultimateCooldown,Weapon:weaponLevel,Nut:shieldBuff,Speed:speedBuff,Pepper:pepperBuff}
 for i:=1;i<192;i++ {if entity[i]!=0 {s.Entities=append(s.Entities,Item{i,entity[i],facing[i],age[i],zombieType[i],hp[i],growth[i],dashWindup[i],pickupType[i],px[i],py[i],velx[i],vely[i],dashRemaining[i],rocket[i],incendiary[i]})}};return s
}
